# Material Admin | SourceBans — требования к хостингу
# Сайт и установщик (install/). PHP-проект, не pip.

# =============================================================================
# ОБЯЗАТЕЛЬНО
# =============================================================================

# Runtime (ветка php-8.3)
PHP >= 8.2
# Целевая / проверенная: PHP 8.3.22
# Для PHP 7.1.33 используй ветку main — см. docs/PHP83.md

# PHP-расширения (обязательные)
php-mysqli          # БД (ADOdb / панель)
php-bcmath          # SteamID / большие числа (проверка в установщике)
php-xml             # XML (проверка в установщике)
php-json            # конфиг, xajax, SEO JSON-LD
php-mbstring        # UTF-8 (имена, банлист, Steam)
php-openssl         # CSRF / токены, HTTPS-клиент
php-curl            # Steam API, антифрод PARSEC, внешние HTTP
php-session         # сессии установщика и панели
php-filter          # валидация входных данных
php-hash            # пароли / sha

# GMP или 64-bit PHP (одно из двух — проверка установщика)
php-gmp             # либо 64-битный PHP (amd64)

# Загрузка файлов
file_uploads = On   # php.ini — демки, иконки, картинки карт

# База данных
MySQL >= 5.0
# Рекомендуется MySQL 5.5+ / MariaDB 10.x
# Кодировка: utf8 / utf8mb4
# Права пользователя БД: SELECT, INSERT, UPDATE, DELETE, CREATE, ALTER, INDEX, DROP

# Веб-сервер
Apache 2.4+ (mod_rewrite желателен)  или  Nginx + PHP-FPM
# Document root = корень панели (где index.php)

# =============================================================================
# РЕКОМЕНДУЕТСЯ
# =============================================================================

php-gd              # генерация images/og-cover.jpg установщиком (SEO)
php-pdo             # data/db.php (совместимость / будущий фреймворк)
php-pdo_mysql
allow_url_fopen = On  # запасной путь к Steam API без curl
# Исходящий UDP/TCP к игровым серверам (query + RCON, обычно порт 27015)

# =============================================================================
# ВНЕШНИЕ СЕРВИСЫ (не ставятся через пакеты)
# =============================================================================

# Steam Web API key → STEAMAPIKEY в config.php (аватарки, профили)
# https://steamcommunity.com/dev/apikey

# Антифрод / LinkedAccounts (PARSEC) — прописывает установщик:
# PARSEC_API_PLAYER_URL = https://api.sibnet-software.ru/api/player/
# Хост должен уметь ходить наружу по HTTPS на api.sibnet-software.ru

# =============================================================================
# ПРАВА НА ЗАПИСЬ (установщик + работа сайта)
# =============================================================================

# Обязательно writable:
demos/
themes_c/
config.php          # или writable корень сайта, чтобы создать файл
data/               # data/db.php, служебное

# Желательно writable:
images/game/
images/maps/
images/             # og-cover.jpg
sitemap.xml
robots.txt
# корень сайта — для одноразового remove_setup.php после установки

# =============================================================================
# УСТАНОВКА (кратко)
# =============================================================================

# 1. Залить файлы в document root
# 2. Создать пустую БД MySQL/MariaDB
# 3. Открыть /install/
# 4. Пройти шаги: лицензия → БД → требования → таблицы → админ
# 5. Удалить install/ (кнопка на финише / remove_setup.php)
# 6. Прописать STEAMAPIKEY в config.php при необходимости

# Проверка без браузера (опционально):
