# Что нужно хостингу

Это не pip и не Python. Файл для людей и для установщика: какие PHP и MySQL нужны панели.

# Обязательно

# Runtime (ветка php-8.3)
PHP >= 8.2
# Лучше PHP 8.3
# Для PHP 7.1 — ветка main, см. docs/PHP83.md

# PHP-расширения
php-mysqli
php-bcmath
php-xml
php-json
php-mbstring
php-openssl
php-curl
php-session
php-filter
php-hash

# Либо GMP, либо 64-битный PHP (установщик это проверяет)
php-gmp

# Загрузка файлов
file_uploads = On

# База
MySQL >= 5.0
# Лучше MySQL 5.5+ или MariaDB 10.x, кодировка utf8mb4
# Права: SELECT, INSERT, UPDATE, DELETE, CREATE, ALTER, INDEX, DROP

# Веб-сервер
Apache 2.4+ (mod_rewrite) или Nginx + PHP-FPM
# Корень сайта — папка с index.php

# Желательно

php-gd
php-pdo
php-pdo_mysql
allow_url_fopen = On
# Исходящий UDP/TCP к игровым серверам (query и RCON, обычно 27015)

# Внешние сервисы

# Ключ Steam Web API → STEAMAPIKEY в config.php
# https://steamcommunity.com/dev/apikey

# Антифрод PARSEC (прописывает установщик):
# PARSEC_API_PLAYER_URL = https://api.sibnet-software.ru/api/player/
# Серверу нужен исходящий HTTPS на api.sibnet-software.ru

# Куда писать на диск

# Обязательно:
demos/
config.php
data/

# Желательно:
images/game/
images/maps/
images/
sitemap.xml
robots.txt

# Установка коротко

# 1. Залить файлы в корень сайта
# 2. Создать пустую БД
# 3. Открыть /install/
# 4. Лицензия → база → проверка → таблицы → админ
# 5. Удалить install/
# 6. При необходимости прописать STEAMAPIKEY

# php install/dry_run.php
